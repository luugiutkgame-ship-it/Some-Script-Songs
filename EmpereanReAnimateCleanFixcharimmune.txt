function Reanimate()
	--====================================================
	-- ANTI-DOUBLE EXECUTION & SERVICE CLONING
	--====================================================
	local genv = (getgenv and getgenv()) or _G
	if genv._EMP_REANIMATE_RUNNING then
		warn("[REANIMATE]: Script đang chạy rồi!")
		return
	end
	genv._EMP_REANIMATE_RUNNING = true

	local function SafeGetService(serviceName)
		local success, service = pcall(function() return game:GetService(serviceName) end)
		if success and service then
			return (cloneref and cloneref(service)) or service
		end
		return nil
	end

	local OptionsAccessories = nil
	local OptionsRigTransparency = 1
	local OptionsRigR15 = nil
	local OptionsRigSetHumanoidDescription = nil
	local OptionsRigSetCharacter = nil
	local OptionsTeleportXandZoffset = nil
	local OptionsTeleportYoffset = nil
	local OptionsDisableScripts = nil
	local OptionsDisableScreenGUIs = nil
	local OptionsFlingEnabled = nil
	local OptionsFlingToolFling = nil
	local OptionsFlingTimeout = nil
	local OptionsFlingVelocity = nil
	local OptionsFlingHighlightTargets = nil
	local OptionsAutoFix = true

	local BindableEvent = nil
	local Boolean = false
	local Humanoid = nil
	local Rig = nil
	local RigHumanoid = nil
	local RigHumanoidRootPart = nil
	local Time = nil
	local DeltaTime = nil
	local LastTime = nil
	local LastValidPosition = nil
	local IsFixingChar = false

	local Attachments = { }
	local BaseParts = { }
	local Blacklist = { }
	local Enableds = { }
	local Highlights = { }
	local RBXScriptConnections = { }
	local RigAccessories = { }
	local Tables = { }
	local Targets = { }

	local CFrame = CFrame
	local CFrameAngles = CFrame.Angles
	local CFrameidentity = CFrame.identity
	local CFramenew = CFrame.new

	local Enum = Enum
	local HumanoidStateType = Enum.HumanoidStateType
	local Physics = HumanoidStateType.Physics
	local Running = HumanoidStateType.Running
	local Track = Enum.CameraType.Track
	local UserInputType = Enum.UserInputType
	local MouseButton1 = UserInputType.MouseButton1
	local Touch = UserInputType.Touch

	local game = game
	local Clone = game.Clone
	local Destroy = game.Destroy
	local FindFirstAncestorOfClass = game.FindFirstAncestorOfClass
	local FindFirstChildOfClass = game.FindFirstChildOfClass
	local GetPropertyChangedSignal = game.GetPropertyChangedSignal
	local GetChildren = game.GetChildren
	local GetDescendants = game.GetDescendants
	local IsA = game.IsA
	
	local Players = SafeGetService("Players")
	local LocalPlayer = Players and Players.LocalPlayer
	if not LocalPlayer then return end

	local Mouse = LocalPlayer:GetMouse()
	local RunService = SafeGetService("RunService")
	local PostSimulation = RunService.PostSimulation
	local PreRender = RunService.PreRender
	local PreSimulation = RunService.PreSimulation
	local Connect = PostSimulation.Connect
	
	local function Disconnect(Connection)
		if Connection then Connection:Disconnect() end
	end
	
	local Wait = PostSimulation.Wait
	local StarterGui = SafeGetService("StarterGui")
	local UserInputService = SafeGetService("UserInputService")
	local IsMouseButtonPressed = UserInputService and UserInputService.IsMouseButtonPressed
	local Workspace = SafeGetService("Workspace")
	local CurrentCamera = Workspace and Workspace.CurrentCamera

	local Instancenew = Instance.new
	local mathabs = math.abs
	local mathrandom = math.random
	local mathsin = math.sin
	local osclock = os.clock
	local pairs = pairs
	local pcall = pcall
	local script = script
	local stringfind = string.find
	local tableclear = table.clear
	
	local tablefind = (table and table.find) or function(tbl, val)
		for i, v in pairs(tbl) do if v == val then return i end end
		return nil
	end

	local tableinsert = table.insert
	local taskspawn = task.spawn
	local taskwait = task.wait
	local typeof = typeof

	local Vector3 = Vector3
	local Vector3new = Vector3.new
	local Vector3zero = Vector3.zero
	local CameraCFrame = CFrameidentity

	local sethiddenproperty = sethiddenproperty or function() end

	-- BẢO VỆ PHÒNG CHỐNG VOID TOÀN CỤC
	pcall(function()
		sethiddenproperty(Workspace, "FallenPartsDestroyHeight", -20000)
		Workspace.FallenPartsDestroyHeight = -20000
	end)

	local Stop

	--====================================================
	-- LẤY ĐIỂM HỒI SINH (SPAWNLOCATION) TRONG WORKSPACE
	--====================================================
	local function GetSpawnCFrame()
		local spawnCandidates = {}
		for _, v in pairs(GetDescendants(Workspace)) do
			if IsA(v, "SpawnLocation") and v.Enabled then
				tableinsert(spawnCandidates, v)
			end
		end

		if #spawnCandidates > 0 then
			local selectedSpawn = spawnCandidates[mathrandom(1, #spawnCandidates)]
			return selectedSpawn.CFrame + Vector3new(0, 5, 0)
		end

		return CFramenew(0, 10, 0)
	end

	--====================================================
	-- AUTO-FIX CHAR SYSTEM
	--====================================================
	local function TriggerAutoFix(reason)
		if IsFixingChar or not OptionsAutoFix then return end
		IsFixingChar = true
		warn("[AUTO-FIX CHAR]: Phát hiện sự cố (" .. tostring(reason) .. ")! Đang phục hồi...")

		if Stop then Stop() end
		taskwait(0.2)
		taskspawn(Reanimate)
	end

	--====================================================
	-- HỆ THỐNG ÉP TÀNG HÌNH RIG MULTIPLIERGUY HOÀN TOÀN
	--====================================================
	local function ForceHideInstance(inst)
		if IsA(inst, "BasePart") or IsA(inst, "Decal") or IsA(inst, "Texture") then
			inst.Transparency = 1
			inst.LocalTransparencyModifier = 1
		elseif IsA(inst, "ParticleEmitter") or IsA(inst, "Fire") or IsA(inst, "Sparkles") or IsA(inst, "Light") then
			inst.Enabled = false
		elseif IsA(inst, "SelectionBox") or IsA(inst, "BoxHandleAdornment") then
			inst.Visible = false
		end
	end

	local function FullyHideRig(rigModel)
		if not rigModel then return end
		for _, item in pairs(GetDescendants(rigModel)) do
			ForceHideInstance(item)
		end
	end

	--====================================================
	-- SAFE RIG CREATOR
	--====================================================
	local function SafeCreateRig(isR15)
		local success, rigModel = pcall(function()
			return Players:CreateHumanoidModelFromUserId(isR15 and 5532894300 or 5532891747)
		end)

		if success and rigModel then
			return rigModel
		end

		local char = LocalPlayer.Character
		if char then
			char.Archivable = true
			local rigClone = char:Clone()
			char.Archivable = false

			for _, item in pairs(rigClone:GetChildren()) do
				if item:IsA("Accessory") or item:IsA("Script") or item:IsA("LocalScript") then
					item:Destroy()
				end
			end

			return rigClone
		end
		return nil
	end

	local function BreakJoints(Parent)
		if not Parent then return end
		for _, Instance in pairs(GetDescendants(Parent)) do
			if IsA(Instance, "JointInstance") or IsA(Instance, "Constraint") then
				Destroy(Instance)
			end
		end
	end

	local function HasJoint(Part0)
		if not Part0 then return false end
		for _, Instance in pairs(GetDescendants(Part0)) do
			if IsA(Instance, "JointInstance") or IsA(Instance, "Constraint") then return true end
		end
		return false
	end

	local function IsValidPart(Part)
		return Part and Part.Parent and Part:IsDescendantOf(Workspace)
	end

	--====================================================
	-- REAL-TIME HUMANOID GUARD
	--====================================================
	local function AutoGuardHumanoid(CharModel, TargetType)
		if not CharModel then return end
		
		local function EnsureHumanoid()
			if not CharModel or not CharModel.Parent then return nil end
			local ExistingHum = CharModel:FindFirstChildOfClass("Humanoid")
			if not ExistingHum then
				ExistingHum = Instancenew("Humanoid")
				ExistingHum.Name = "Humanoid"
				ExistingHum.Parent = CharModel
			end
			return ExistingHum
		end

		local CurrentHum = EnsureHumanoid()
		if TargetType == "LocalPlayer" then
			Humanoid = CurrentHum
		elseif TargetType == "Rig" then
			RigHumanoid = CurrentHum
		end

		tableinsert(RBXScriptConnections, Connect(CharModel.ChildRemoved, function(child)
			if child:IsA("Humanoid") or child.Name == "Humanoid" then
				taskspawn(function()
					taskwait(0.05)
					if CharModel and CharModel.Parent then
						local NewHum = EnsureHumanoid()
						if TargetType == "LocalPlayer" then Humanoid = NewHum
						elseif TargetType == "Rig" then RigHumanoid = NewHum end
					else
						if TargetType == "Rig" and OptionsAutoFix then
							TriggerAutoFix("Rig Humanoid Bị Xóa Hoàn Toàn")
						end
					end
				end)
			end
		end))
	end

	--====================================================
	-- CAMERA CONTROL
	--====================================================
	local function CameraSubject()
		if RigHumanoid and RigHumanoid.Parent then CurrentCamera.CameraSubject = RigHumanoid end
		Wait(PreRender)
		CurrentCamera.CFrame = CameraCFrame
	end

	local function CameraType()
		if CurrentCamera.CameraType ~= Track then CurrentCamera.CameraType = Track end
	end

	local function Camera()
		local Cam = Workspace.CurrentCamera
		if Cam then
			CameraCFrame = Cam.CFrame
			CurrentCamera = Cam
			CameraSubject()
			CameraType()
			tableinsert(RBXScriptConnections, Connect(GetPropertyChangedSignal(Cam, "CameraSubject"), CameraSubject))
			tableinsert(RBXScriptConnections, Connect(GetPropertyChangedSignal(Cam, "CameraType"), CameraType))
		end
	end

	local function FindFirstChildOfClassAndName(Parent, ClassName, Name)
		if not Parent then return nil end
		for _, Instance in pairs(GetChildren(Parent)) do
			if IsA(Instance, ClassName) and Instance.Name == Name then return Instance end
		end
	end

	local function WaitForChildOfClassAndName(Parent, ClassName, Name, Timeout)
		Timeout = Timeout or 2
		local StartTime = osclock()
		local Instance = FindFirstChildOfClassAndName(Parent, ClassName, Name)
		while not Instance and typeof(Parent) == "Instance" and (osclock() - StartTime) < Timeout do
			Instance = FindFirstChildOfClassAndName(Parent, ClassName, Name)
			taskwait(0.05)
		end
		return Instance
	end

	local function Invisible(Instance)
		ForceHideInstance(Instance)
	end

	local function DescendantAdded(Instance)
		if IsA(Instance, "Attachment") then
			local Handle = Instance.Parent
			if IsA(Handle, "BasePart") then
				local AttachmentsAttachment = Attachments[Instance.Name]
				if AttachmentsAttachment then
					local MeshId, TextureId = "", ""
					if IsA(Handle, "MeshPart") then
						MeshId, TextureId = Handle.MeshId, Handle.TextureID
					else
						local SpecialMesh = FindFirstChildOfClass(Handle, "SpecialMesh")
						if SpecialMesh then MeshId, TextureId = SpecialMesh.MeshId, SpecialMesh.TextureId end
					end

					for _, Table in pairs(OptionsAccessories) do
						if stringfind(MeshId, Table.MeshId or "") and stringfind(TextureId, Table.TextureId or "") then
							local TargetPart = FindFirstChildOfClassAndName(Rig, "BasePart", Table.Name)
							local AlternativeName = Table.AlternativeName
							local AlternativeInstance = false

							if not TargetPart and AlternativeName then
								TargetPart = FindFirstChildOfClassAndName(Rig, "BasePart", AlternativeName)
								AlternativeInstance = true
							end

							if TargetPart and not tablefind(Blacklist, TargetPart) then
								if Table.Blacklist then tableinsert(Blacklist, TargetPart) end
								BreakJoints(Handle)
								tableinsert(Tables, { Part0 = Handle, Part1 = TargetPart, CFrame = AlternativeInstance and Table.AllowAlternativeCFrame and Table.AlternativeCFrame or Table.CoordinateFrame, LastPosition = TargetPart.Position })
								return
							end
						end
					end

					for _, Table in pairs(RigAccessories) do
						local TableHandle = Table.Handle
						if typeof(TableHandle) == "Instance" and Table.MeshId == MeshId and Table.TextureId == TextureId then
							BreakJoints(Handle)
							tableinsert(Tables, { Part0 = Handle, Part1 = TableHandle, LastPosition = TableHandle.Position })
							return
						end
					end

					local Accessory = Handle.Parent
					if IsA(Accessory, "Accessory") then
						local AccessoryClone = Instancenew("Accessory")
						AccessoryClone.Name = Accessory.Name
						local HandleClone = Clone(Handle)
						Invisible(HandleClone)
						BreakJoints(HandleClone)
						HandleClone.Parent = AccessoryClone

						local Weld = Instancenew("Weld")
						Weld.Name = "AccessoryWeld"
						Weld.C0 = Instance.CFrame
						Weld.C1 = AttachmentsAttachment.CFrame
						Weld.Part0 = HandleClone
						Weld.Part1 = AttachmentsAttachment.Parent
						Weld.Parent = HandleClone

						tableinsert(RigAccessories, { Handle = HandleClone, MeshId = MeshId, TextureId = TextureId })
						tableinsert(Tables, { Part0 = Handle, Part1 = HandleClone, LastPosition = HandleClone.Position })
						AccessoryClone.Parent = Rig
					end
				end
			end
		elseif IsA(Instance, "BasePart") then
			Instance.CanQuery = false
			tableinsert(BaseParts, Instance)
		end
	end

	local function SafeDisableScript(Instance)
		if not Instance or Instance == script then return end
		if IsA(Instance, "LocalScript") or IsA(Instance, "Script") then
			local Char = LocalPlayer.Character
			if Char and Instance:IsDescendantOf(Char) then
				pcall(function()
					Enableds[Instance] = Instance.Enabled
					Instance.Enabled = false
				end)
			end
		elseif OptionsDisableScreenGUIs and IsA(Instance, "ScreenGui") then
			pcall(function()
				Enableds[Instance] = Instance.Enabled
				Instance.Enabled = false
			end)
		end
	end

	local function gameDescendantAdded(Instance)
		if OptionsDisableScripts or OptionsDisableScreenGUIs then
			SafeDisableScript(Instance)
		end
	end

	local function ApplyDescriptionRig()
		if not Humanoid or not Humanoid.Parent or not RigHumanoid or not RigHumanoid.Parent then return end
		pcall(function()
			local Description = RigHumanoid:GetAppliedDescription()
			Description.HatAccessory = ""
			Description.BackAccessory = ""
			Description.FaceAccessory = ""
			Description.HairAccessory = ""
			Description.NeckAccessory = ""
			Description.FrontAccessory = ""
			Description.WaistAccessory = ""
			Description.ShouldersAccessory = ""
			RigHumanoid:ApplyDescription(Description)
		end)

		for _, Instance in pairs(GetDescendants(Rig)) do Invisible(Instance) end
	end

	local function SetCharacter()
		taskwait(0.1)
		LocalPlayer.Character = Rig
	end

	local Tools = {}
	local function CharacterAdded()
		local Character = LocalPlayer.Character
		if Character and Character ~= Rig then
			AutoGuardHumanoid(Character, "LocalPlayer")

			if OptionsDisableScripts then
				for _, child in pairs(GetDescendants(Character)) do
					SafeDisableScript(child)
				end
			end

			if OptionsFlingToolFling then
				for _, Backpack in pairs(GetChildren(LocalPlayer)) do
					if IsA(Backpack, "Backpack") then
						for _, Instance in pairs(GetChildren(Backpack)) do
							if IsA(Instance, "Tool") then
								Tools[Instance] = FindFirstChildOfClassAndName(Instance, "BasePart", "Handle")	
								Instance.Parent = Character
							end
						end
					end
				end
			end

			if OptionsRigSetCharacter then taskspawn(SetCharacter) end

			tableclear(BaseParts)
			tableclear(Blacklist)
			tableclear(Tables)

			if CurrentCamera then CameraCFrame = CurrentCamera.CFrame end
			for _, Instance in pairs(GetDescendants(Character)) do DescendantAdded(Instance) end
			tableinsert(RBXScriptConnections, Connect(Character.DescendantAdded, DescendantAdded))

			local HumanoidRootPart = WaitForChildOfClassAndName(Character, "BasePart", "HumanoidRootPart", 2)

			if Boolean then
				Camera()
				if HumanoidRootPart and RigHumanoidRootPart then
					if HumanoidRootPart.Position.Y > 0 and HumanoidRootPart.Position.Y < 50000 then
						RigHumanoidRootPart.CFrame = HumanoidRootPart.CFrame
					else
						-- NẾU BỊ VOID KHI SPAWN, DÙNG SPAWNLOCATION
						RigHumanoidRootPart.CFrame = GetSpawnCFrame()
					end
					Boolean = false
				end

				if OptionsRigSetHumanoidDescription then pcall(ApplyDescriptionRig) end
			end

			if HumanoidRootPart then
				for Index, Table in pairs(Targets) do
					if not HumanoidRootPart then break end
					if Humanoid and Humanoid.Parent then pcall(function() Humanoid:ChangeState(Physics) end) end

					local Target = Table.Target
					local Timeout = Time + OptionsFlingTimeout
					local LastPosition = Target.Position

					while Target and HumanoidRootPart do
						if Time > Timeout then break end
						local Position = Target.Position
						local LinearVelocity = (Position - LastPosition) / DeltaTime

						if LinearVelocity.Magnitude > 50 then break end
						LastPosition = Position

						HumanoidRootPart.AssemblyAngularVelocity = OptionsFlingVelocity
						HumanoidRootPart.AssemblyLinearVelocity = OptionsFlingVelocity
						HumanoidRootPart.CFrame = Target.CFrame * CFramenew(0, 0, 4 * mathsin(Time * 30)) * CFrameAngles(mathrandom(-360, 360), mathrandom(-360, 360), mathrandom(-360, 360)) + (LinearVelocity * 0.5) 
						taskwait()
					end

					local Highlight = Table.Highlight
					if Highlight then Destroy(Highlight) end
					Targets[Index] = nil
				end

				if Humanoid and Humanoid.Parent then pcall(function() Humanoid:ChangeState(Running) end) end

				if RigHumanoidRootPart then
					HumanoidRootPart.AssemblyAngularVelocity = Vector3zero
					HumanoidRootPart.AssemblyLinearVelocity = Vector3zero
					HumanoidRootPart.CFrame = RigHumanoidRootPart.CFrame * CFramenew(0, -18, 0)
				end
			end
			
			taskwait(0.2)
			if Character then BreakJoints(Character) end
		end
	end

	--====================================================
	-- RUNTIME HOOKS & VOID / HIDING MONITORING
	--====================================================
	local function PreRenderConnect()
		-- ÉP TÀNG HÌNH RIG MULTIPLIERGUY TRONG MỖI FRAME HIỂN THỊ
		if Rig then
			FullyHideRig(Rig)
		end
	end

	local function PostSimulationConnect()
		pcall(sethiddenproperty, LocalPlayer, "SimulationRadius", 2147483647)

		Time = osclock()
		DeltaTime = Time - (LastTime or Time)
		LastTime = Time

		-- AUTO-FIX DETECTION
		if OptionsAutoFix and not IsFixingChar then
			if not Rig or not Rig.Parent or not Rig:IsDescendantOf(Workspace) then
				TriggerAutoFix("Rig Bị Unparent Hoặc Xóa")
				return
			end
			if not RigHumanoidRootPart or not RigHumanoidRootPart.Parent then
				TriggerAutoFix("Rig HumanoidRootPart Bị Mất")
				return
			end
			if RigHumanoid and RigHumanoid.Health <= 0 then
				TriggerAutoFix("Rig Humanoid Bị Chết")
				return
			end
		end

		local Integer = 25.05
		local Vector3Offset = Vector3new(0, 0, 0.002 * mathsin(Time * 25))

		for _, Table in pairs(Tables) do
			local Part0 = Table.Part0
			local Part1 = Table.Part1

			if IsValidPart(Part0) and IsValidPart(Part1) and not HasJoint(Part0) and Part0.ReceiveAge <= 0.1 then
				Part0.AssemblyAngularVelocity = Vector3zero
				local Position = Part1.Position
				local LinearVelocity = ((Table.LastPosition - Position) / DeltaTime) * Integer
				Table.LastPosition = Position

				Part0.AssemblyLinearVelocity = Vector3new(LinearVelocity.X, Integer, LinearVelocity.Z)
				Part0.CFrame = Part1.CFrame * (Table.CFrame or CFrameidentity) + Vector3Offset
			end
		end

		local Hit = Mouse and Mouse.Hit
		local Holding = UserInputService and IsMouseButtonPressed(UserInputService, MouseButton1)
		
		for Tool, BasePart in pairs(Tools) do
			if IsValidPart(BasePart) and BasePart.ReceiveAge <= 0.1 and not HasJoint(BasePart) then
				BasePart.CanCollide = false
				BasePart.AssemblyAngularVelocity = OptionsFlingVelocity
				BasePart.AssemblyLinearVelocity = OptionsFlingVelocity
				
				if Holding and Hit then
					BasePart.CFrame = Hit
				elseif RigHumanoidRootPart then
					BasePart.CFrame = RigHumanoidRootPart.CFrame * CFramenew(0, OptionsTeleportYoffset, 0) * CFrameAngles(mathrandom(-360, 360), mathrandom(-360, 360), mathrandom(-360, 360))
				else
					BasePart.CFrame = CFrameidentity
				end
			end
		end

		if RigHumanoid and RigHumanoid.Parent and Humanoid and Humanoid.Parent then
			pcall(function()
				RigHumanoid.Jump = Humanoid.Jump
				RigHumanoid:Move(Humanoid.MoveDirection)
			end)
		end

		--====================================================
		-- CHỈ KHI RƠI XUỐNG VOID (Y < -100): SPAWN LẠI TẠI SPAWNLOCATION
		--====================================================
		if RigHumanoidRootPart then
			if RigHumanoidRootPart.Position.Y < -100 then
				local SpawnCFrame = GetSpawnCFrame()
				RigHumanoidRootPart.CFrame = SpawnCFrame
				RigHumanoidRootPart.AssemblyLinearVelocity = Vector3zero
				RigHumanoidRootPart.AssemblyAngularVelocity = Vector3zero
				
				-- CĂN CHỈNH LẠI NHÂN VẬT THẬT VÀ TỰ ĐỘNG KHÔI PHỤC VỊ TRÍ HAT
				local RealChar = LocalPlayer.Character
				if RealChar and RealChar ~= Rig then
					local RealHRP = FindFirstChildOfClassAndName(RealChar, "BasePart", "HumanoidRootPart")
					if RealHRP then
						RealHRP.CFrame = SpawnCFrame * CFramenew(0, -18, 0)
						RealHRP.AssemblyLinearVelocity = Vector3new(0, 25.05, 0)
						RealHRP.AssemblyAngularVelocity = Vector3zero
					end
				end
			end

			-- BẢO VỆ CHỐNG RƠI QUÁ VẬN TỐC CHO RIG
			if Rig then
				for _, Part in pairs(GetDescendants(Rig)) do
					if IsA(Part, "BasePart") then
						if Part.AssemblyLinearVelocity.Magnitude > 250 or Part.AssemblyAngularVelocity.Magnitude > 250 then
							Part.AssemblyLinearVelocity = Vector3zero
							Part.AssemblyAngularVelocity = Vector3zero
						end
					end
				end
			end

			local RealChar = LocalPlayer.Character
			if RealChar and RealChar ~= Rig then
				local RealHRP = FindFirstChildOfClassAndName(RealChar, "BasePart", "HumanoidRootPart")
				if RealHRP and RealHRP.Position.Y >= -100 then
					RealHRP.CFrame = RigHumanoidRootPart.CFrame * CFramenew(0, -18, 0)
					RealHRP.AssemblyLinearVelocity = Vector3new(0, 25.05, 0)
					RealHRP.AssemblyAngularVelocity = Vector3zero
				end
				
				for _, BPart in pairs(BaseParts) do
					if IsValidPart(BPart) and BPart.Name ~= "HumanoidRootPart" and BPart.Parent and not IsA(BPart.Parent, "Accessory") then
						if BPart.AssemblyLinearVelocity.Magnitude > 250 or BPart.AssemblyAngularVelocity.Magnitude > 250 then
							BPart.AssemblyLinearVelocity = Vector3zero
							BPart.AssemblyAngularVelocity = Vector3zero
						end
						
						if not HasJoint(BPart) then
							BPart.CFrame = RigHumanoidRootPart.CFrame * CFramenew(0, -18, 0)
							BPart.AssemblyLinearVelocity = Vector3new(0, 25.05, 0)
							BPart.AssemblyAngularVelocity = Vector3zero
						end
					end
				end
			end
		end
	end

	local function PreSimulationConnect()
		for _, BasePart in pairs(BaseParts) do BasePart.CanCollide = false end
		if Rig then
			for _, Part in pairs(GetDescendants(Rig)) do
				if IsA(Part, "BasePart") then Part.CanCollide = false end
			end
		end
		for _, Player in pairs(GetChildren(Players)) do
			if Player ~= LocalPlayer and Player.Character then
				for _, Part in pairs(GetDescendants(Player.Character)) do
					if IsA(Part, "BasePart") then Part.CanCollide = false end
				end
			end
		end
	end

	local function Fling(Target)
		if typeof(Target) == "Instance" then
			if IsA(Target, "Humanoid") then Target = Target.Parent end
			if IsA(Target.Parent, "Accessory") then Target = FindFirstAncestorOfClass(Target, "Model") end
			if IsA(Target, "Model") then Target = FindFirstChildOfClassAndName(Target, "BasePart", "HumanoidRootPart") end
			if IsA(Target, "BasePart") then
				for _, Table in pairs(Targets) do
					if Table.Target == Target then return end
				end

				local Parent = Target.Parent
				local Highlight = nil
				if OptionsFlingHighlightTargets then
					Highlight = Instancenew("Highlight")
					Highlight.Adornee = Parent
					Highlight.Parent = Parent
					tableinsert(Highlights, Highlight)
				end
				tableinsert(Targets, {Highlight = Highlight, Target = Target})
			end
		end
	end

	local function InputBegan(InputObject)
		local UserInputTypeVal = InputObject.UserInputType
		if UserInputTypeVal == MouseButton1 or UserInputTypeVal == Touch then
			local Target = Mouse and Mouse.Target
			if Target and not Target.Anchored then
				local Model = Target.Parent
				if IsA(Model, "Model") and FindFirstChildOfClass(Model, "Humanoid") then
					local HumanoidRootPart = FindFirstChildOfClassAndName(Model, "BasePart", "HumanoidRootPart")
					if HumanoidRootPart then Fling(HumanoidRootPart) end
				else
					Fling(Target)
				end
			end
		end
	end

	Stop = function()
		for _, RBXScriptConnection in pairs(RBXScriptConnections) do Disconnect(RBXScriptConnection) end
		for _, Highlight in pairs(Highlights) do Destroy(Highlight) end
		for Instance, BooleanVal in pairs(Enableds) do 
			pcall(function() Instance.Enabled = BooleanVal end)
		end

		tableclear(Attachments)
		tableclear(BaseParts)
		tableclear(Enableds)
		tableclear(Highlights)
		tableclear(RBXScriptConnections)
		tableclear(Tables)
		tableclear(Targets)

		if Rig then Destroy(Rig) end
		if BindableEvent then Destroy(BindableEvent) end
		genv._EMP_REANIMATE_RUNNING = nil
	end
